Идея: подсчитать количество всех индексов, которые бывают при запросе. Подставить под наиболее частые индексы наибольшие значения.
Но как подсчитать встречающиеся значения? Если запросов может быть 10^5 и интервалы могут быть шириной 10^5  
# Код
```python
def solution(n, q, a, queries):

    # your code

    #Наибольшие элементы будут повторяться наибольшее количество раз

    counts = [0] * n

    for l, r in queries:

        counts[l - 1] += 1

        # после позиции r, влияние этого запроса прекращается    

        if r < n:

            counts[r] -= 1    

    # преобразуем разностный массив в префикс сумм    

    for i in range(1, n):

        counts[i] += counts[i - 1] # то есть мы отметили левые границы отрезков, и

    print(counts)

  
  

    a.sort(reverse=True)

    counts.sort(reverse=True)

  

    max_joy = 0

    for i in range(n):

        if counts[i] == 0:

            break

        max_joy += a[i] * counts[i]

    return max_joy
```
